cmake_minimum_required(VERSION 3.24)
project(OpenExrReference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../Scripts/cmake/OpenEXR.cmake")
foreach(tool codec_oracle file_oracle)
  add_executable(${tool} ${tool}.cpp)
  target_link_libraries(${tool} PRIVATE OpenEXR::OpenEXR)
  target_include_directories(${tool} PRIVATE "${openexr_SOURCE_DIR}/src/lib/OpenEXR")
endforeach()
