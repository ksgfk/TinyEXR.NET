cmake_minimum_required(VERSION 3.20)
project(TinyExrReference C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
get_filename_component(UPSTREAM "${CMAKE_CURRENT_SOURCE_DIR}/../../TinyEXR.Native/tinyexr" ABSOLUTE)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${UPSTREAM}" OUTPUT_VARIABLE REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT REV STREQUAL "1b106618644dbf8a0935c2348ba51a2d863dd7c2")
  message(FATAL_ERROR "Unexpected upstream reference revision: ${REV}")
endif()
file(GLOB SOURCES CONFIGURE_DEPENDS "${UPSTREAM}/src/*.c")
list(FILTER SOURCES EXCLUDE REGEX "/exr_zstd\\.c$")
add_library(tinyexr_reference STATIC ${SOURCES})
target_include_directories(tinyexr_reference PUBLIC "${UPSTREAM}/include" "${UPSTREAM}/src")
target_compile_definitions(tinyexr_reference PUBLIC EXR_NO_ZSTD _CRT_SECURE_NO_WARNINGS)
if(NOT WIN32)
  target_link_libraries(tinyexr_reference PUBLIC m)
endif()
foreach(tool parse_harness encode_harness compare_exr)
  add_executable(${tool} "${UPSTREAM}/test/v3/${tool}.c")
  target_link_libraries(${tool} PRIVATE tinyexr_reference)
endforeach()
add_executable(codec_oracle codec_oracle.c)
target_link_libraries(codec_oracle PRIVATE tinyexr_reference)
add_executable(file_oracle file_oracle.c)
target_link_libraries(file_oracle PRIVATE tinyexr_reference)
